class BalancedBrackets:
    def __init__(self, expr):
        self.stack = []
        self.left_brackets = ["(", "{", "["]
        self.right_brackets = [")", "}", "]"]
        self.expr = expr

    def clear_expr(self):
        self.expr = "".join(c for c in self.expr if c in self.left_brackets or c in self.right_brackets)

    def check_balanced_brackets(self):
        self.clear_expr()
        self.stack = []
        for c in self.expr:
            if c in self.left_brackets:
                self.stack.append(c)
            else:
                if not self.stack:
                    return False
                top = self.stack.pop()
                if self.left_brackets.index(top) != self.right_brackets.index(c):
                    return False
        return len(self.stack) == 0
