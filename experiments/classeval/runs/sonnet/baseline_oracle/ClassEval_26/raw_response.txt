```python
import csv

class CSVProcessor:
    def __init__(self):
        pass

    def read_csv(self, file_name):
        with open(file_name, 'r') as f:
            reader = csv.reader(f)
            rows = list(reader)
        title = rows[0]
        data = rows[1:]
        return title, data

    def write_csv(self, data, file_name):
        try:
            with open(file_name, 'w', newline='') as f:
                writer = csv.writer(f)
                writer.writerows(data)
            return 1
        except:
            return 0

    def process_csv_data(self, N, save_file_name):
        try:
            title, data = self.read_csv(save_file_name)
            new_data = [[row[N].upper()] for row in data]
            base = save_file_name.replace('.csv', '')
            new_file = base + '_process.csv'
            return self.write_csv([title] + new_data, new_file)
        except:
            return 0
```
